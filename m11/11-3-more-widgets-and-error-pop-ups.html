Handling bad input with messagebox

Combine try/except from Module 8 with a pop-up so your app never crashes:

import tkinter as tk
from tkinter import messagebox

def convert():
    try:
        f = float(temp_entry.get())
    except ValueError:
        messagebox.showerror("Invalid input", "Please enter a number.")
        return
    c = (f - 32) * 5 / 9
    result.config(text=f"{f:.1f} F = {c:.1f} C")

window = tk.Tk()
window.title("Temperature Converter")
tk.Label(window, text="Temperature (F):").grid(row=0, column=0, padx=5, pady=5)
temp_entry = tk.Entry(window)
temp_entry.grid(row=0, column=1, padx=5)
tk.Button(window, text="Convert", command=convert).grid(row=1, column=0, columnspan=2, pady=5)
result = tk.Label(window, text="")
result.grid(row=2, column=0, columnspan=2, pady=5)
window.mainloop()

Checkbuttons and Spinbox

import tkinter as tk

def show_order():
    size = size_spin.get()
    extras = []
    if cheese_var.get():
        extras.append("extra cheese")
    if spicy_var.get():
        extras.append("spicy")
    summary.config(text=f"{size}-inch pizza" + (f" with {', '.join(extras)}" if extras else ""))

window = tk.Tk()
window.title("Pizza Order")
tk.Label(window, text="Size (inches):").pack()
size_spin = tk.Spinbox(window, from_=8, to=18, increment=2, width=5)
size_spin.pack()
cheese_var = tk.BooleanVar()
spicy_var = tk.BooleanVar()
tk.Checkbutton(window, text="Extra cheese", variable=cheese_var).pack()
tk.Checkbutton(window, text="Spicy", variable=spicy_var).pack()
tk.Button(window, text="Order", command=show_order).pack(pady=5)
summary = tk.Label(window, text="")
summary.pack(pady=5)
window.mainloop()

Copying to the clipboard

import tkinter as tk
window = tk.Tk()
def copy_text():
    window.clipboard_clear()
    window.clipboard_append("Copied from Python!")
tk.Button(window, text="Copy", command=copy_text).pack(padx=40, pady=20)
window.mainloop()
